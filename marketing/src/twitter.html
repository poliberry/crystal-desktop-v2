fish-alarm.ts
import { Client, Events } from "@crystal/bot";

const client = new Client();

// Someone mentioned the fish again.
client.on(Events.MessageCreate, async (message) => {
  if (message.fromBot) return;
  if (message.content.includes("fish")) {
    await message.react("🐟");
  }
});
